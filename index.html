// เอา URL ที่ได้จากขั้นตอนที่ 2 มาใส่ในเครื่องหมายคำพูดด้านล่างนี้ const SCRIPT_URL = 'https://script.google.com/macros/s/AKfycbwQCD5pRR0-FHwsSzk26xmqgoNo42mSAS9yWyBKp3eEUxLMwA7zarqwuWKjcd5oq3XyBA/exec'; function submitOrder(tableNumber, cart, totalPrice) { // นำรายการอาหารมารวมเป็นข้อความยาวๆ เช่น "กะเพรา x2, ชาเย็น x1" const orderItemsText = cart.map(item => `${item.name} x${item.quantity}`).join(', '); // เตรียมข้อมูลที่จะส่งไป Google Sheets const data = { table: tableNumber, items: orderItemsText, total: totalPrice }; // เปลี่ยนปุ่มเป็นกำลังโหลด document.getElementById('checkout-button').innerText = 'กำลังส่งออเดอร์...'; // ส่งข้อมูลไปที่ Google Sheets fetch(SCRIPT_URL, { method: 'POST', body: JSON.stringify(data) }) .then(response => response.json()) .then(result => { if (result.result === 'success') { alert('สั่งอาหารสำเร็จ! ข้อมูลถูกส่งไปยังร้านแล้ว'); // ล้างตะกร้าสินค้าและรีเซ็ตหน้าจอ (เขียนโค้ดล้างตะกร้าของคุณตรงนี้) window.location.reload(); } }) .catch(error => { console.error('Error!', error.message); alert('เกิดข้อผิดพลาดในการสั่งอาหาร กรุณาลองใหม่อีกครั้ง'); }); }
